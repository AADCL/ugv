cmake_minimum_required(VERSION 3.0.2)
project(scout_r3live_bringup)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs sensor_msgs geometry_msgs
  message_filters tf2_ros)
find_package(Eigen3 REQUIRED)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
add_executable(project_interface_node src/project_interface.cpp)
target_link_libraries(project_interface_node ${catkin_LIBRARIES})
install(TARGETS project_interface_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
catkin_install_python(PROGRAMS scripts/session.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(FILES scripts/calibration_io.py scripts/runtime_health.py scripts/operation_contract.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
