cmake_minimum_required(VERSION 3.0.2)
project(wheeltec_exploration_driver)
add_compile_options(-std=c++11 -O2)
find_package(catkin REQUIRED COMPONENTS roscpp serial tf std_msgs nav_msgs geometry_msgs sensor_msgs)
catkin_package()
include_directories(include ${catkin_INCLUDE_DIRS})
add_executable(exploration_chassis_node src/wheeltec_robot.cpp src/Quaternion_Solution.cpp)
target_link_libraries(exploration_chassis_node ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_exploration_watchdog test/test_cmd_vel_watchdog.cpp)
  if(TARGET test_exploration_watchdog)
    target_link_libraries(test_exploration_watchdog gtest_main ${catkin_LIBRARIES})
  endif()
endif()
install(TARGETS exploration_chassis_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
